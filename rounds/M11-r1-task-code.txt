    def _entry_text(entry):
        parts = [str(entry.get("command") or "")]
        parts.append(_inline_script(entry.get("args")))
        env = entry.get("env")
        if isinstance(env, dict):
            parts.extend(str(v) for v in env.values())
        return " ".join(parts)             # <<< url 不在其中

    def _filter_suspicious_mcp_servers(servers):
        from hermes_cli.mcp_security import validate_mcp_server_entry
        safe = {}
        for name, cfg in servers.items():
            issues = validate_mcp_server_entry(name, cfg) if isinstance(cfg, dict) else None
            if issues:
                logger.warning("Skipping suspicious MCP server '%s': %s", name, "; ".join(issues))
            else:
                safe[name] = cfg
        return safe


三处结构性缺口:
  1. 检查②③ 被 command 类型门控，只对 shell 解释器生效
  2. url 型条目无 command -> 早退 -> 零检查
  3. _entry_text 不含 url -> IOC 检查也覆盖不到 url
  4. command 指向的脚本正文从不被读取
